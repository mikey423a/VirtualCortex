# Prefer an existing JUCE checkout for reproducible/offline Windows builds.
# Supply -DVC_JUCE_SOURCE_DIR=C:/src/JUCE to avoid GitHub access during configure.
set(VC_JUCE_SOURCE_DIR "" CACHE PATH "Path to an existing JUCE source checkout")
option(VC_ALLOW_JUCE_FETCH "Allow CMake to fetch JUCE when not provided locally" OFF)
if(VC_JUCE_SOURCE_DIR)
  if(NOT EXISTS "${VC_JUCE_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "VC_JUCE_SOURCE_DIR must point to the JUCE source root")
  endif()
  add_subdirectory("${VC_JUCE_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/juce-local")
elseif(VC_ALLOW_JUCE_FETCH)
  include(FetchContent)
  FetchContent_Declare(juce GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(juce)
else()
  message(FATAL_ERROR "JUCE is required: set -DVC_JUCE_SOURCE_DIR=<JUCE checkout> or enable VC_ALLOW_JUCE_FETCH")
endif()
juce_add_plugin(VirtualCortex COMPANY_NAME "VirtualCortex" BUNDLE_ID "com.virtualcortex.audio" PLUGIN_MANUFACTURER_CODE VCor PLUGIN_CODE VC01 FORMATS VST3 Standalone PRODUCT_NAME "Virtual Cortex" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
# ASIO is opt-in: it requires the Steinberg ASIO SDK available to JUCE.
# The standalone app's device selector is supplied by JUCE.
option(VC_ENABLE_ASIO "Enable ASIO device support in Windows standalone (requires ASIO SDK)" OFF)
if(VC_ENABLE_ASIO)
  if(NOT WIN32)
    message(FATAL_ERROR "VC_ENABLE_ASIO is only supported for Windows builds")
  endif()
  set(VC_ASIO_SDK_DIR "" CACHE PATH "Steinberg ASIO SDK root")
  if(NOT EXISTS "${VC_ASIO_SDK_DIR}/common/iasiodrv.h")
    message(FATAL_ERROR "VC_ASIO_SDK_DIR must contain common/iasiodrv.h")
  endif()
  target_include_directories(VirtualCortex PUBLIC "${VC_ASIO_SDK_DIR}/common")
  target_compile_definitions(VirtualCortex PUBLIC JUCE_ASIO=1)
endif()
